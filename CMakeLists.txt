 # Some supported Geant4 packages discover C dependencies (e.g. HDF5).
 enable_language(C)

 add_executable(spacerad_worker engine/RadiationWorker.cpp ${_transport} engine/runtime/RadiationRuntime.cpp engine/runtime/BatchRunner.cpp engine/runtime/ShieldingScanRunner.cpp engine/runtime/ResponseCacheRunner.cpp)
 set_target_properties(spacerad_worker PROPERTIES OUTPUT_NAME scdat_radiation_worker AUTOMOC OFF)
 target_include_directories(spacerad_worker PRIVATE ${Geant4_INCLUDE_DIRS})
 target_link_libraries(spacerad_worker PRIVATE spacerad_io ${Geant4_LIBRARIES})
 set(_manifest "")
 foreach(_dataset IN LISTS Geant4_DATASETS)
  get_filename_component(_name "${Geant4_DATASET_${_dataset}_PATH}" NAME)
  string(APPEND _manifest "${Geant4_DATASET_${_dataset}_ENVVAR}|${_name}\n")
 endforeach()
 set(_context "{\"schema\":\"spacerad.worker-build/1\",\"compiled_geant4_version\":\"${Geant4_VERSION}\",\"worker_role\":\"standalone\",\"application_version\":\"${PROJECT_VERSION}\",\"paths\":{\"qt_prefix\":\".\",\"worker_directory\":\".\"}}\n")
 string(SHA256 _context_sha "${_context}")
 string(SHA256 _manifest_sha "${_manifest}")
 target_compile_definitions(spacerad_worker PRIVATE SCDAT_BUILD_CONTEXT_SHA256="${_context_sha}" SCDAT_DATASET_MANIFEST_SHA256="${_manifest_sha}")
 file(GENERATE OUTPUT "$<TARGET_FILE_DIR:spacerad_worker>/radiation-build.json" CONTENT "${_context}" NEWLINE_STYLE UNIX)
 file(GENERATE OUTPUT "$<TARGET_FILE_DIR:spacerad_worker>/radiation-datasets.txt" CONTENT "${_manifest}" NEWLINE_STYLE UNIX)
 install(TARGETS spacerad_worker RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 install(FILES "$<TARGET_FILE_DIR:spacerad_worker>/radiation-build.json" "$<TARGET_FILE_DIR:spacerad_worker>/radiation-datasets.txt" DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
