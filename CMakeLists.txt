cmake_minimum_required(VERSION 3.22)
project(SpaceRad VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(GNUInstallDirs)
option(SPACERAD_BUILD_DESKTOP "Build Qt/VTK desktop and embedding example" ON)
option(SPACERAD_BUILD_WORKER "Build separate Geant4 worker" ON)
if(MSVC)
 add_compile_options(/utf-8 /EHsc)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
add_library(spacerad_json INTERFACE)
target_include_directories(spacerad_json INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann_json/include> $<INSTALL_INTERFACE:include>)
add_library(spacerad_model STATIC engine/RadiationProject.cpp engine/Geometry.cpp engine/AnalyticShapes.cpp engine/Materials.cpp engine/Instances.cpp engine/Voxels.cpp engine/PreviewMeshQuality.cpp engine/MeshSpatialAudit.cpp engine/MeshMaterialShells.cpp)
add_library(SpaceRad::Model ALIAS spacerad_model)
set_target_properties(spacerad_model PROPERTIES EXPORT_NAME Model AUTOMOC OFF)
target_include_directories(spacerad_model PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/engine> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/spacerad/engine> $<INSTALL_INTERFACE:include>)
target_link_libraries(spacerad_model PUBLIC spacerad_json)
file(GLOB _io CONFIGURE_DEPENDS engine/runtime/*.cpp)
list(REMOVE_ITEM _io "${CMAKE_CURRENT_SOURCE_DIR}/engine/runtime/RadiationRuntime.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/engine/runtime/BatchRunner.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/engine/runtime/ShieldingScanRunner.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/engine/runtime/ResponseCacheRunner.cpp")
add_library(spacerad_io STATIC ${_io})
set_target_properties(spacerad_io PROPERTIES AUTOMOC OFF)
target_link_libraries(spacerad_io PUBLIC spacerad_model Qt6::Core)
if(MSVC)
 set_source_files_properties(engine/MeshSpatialAudit.cpp engine/runtime/ConvexSeparation.cpp engine/runtime/StarFanSeparation.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
else()
 set_source_files_properties(engine/MeshSpatialAudit.cpp engine/runtime/ConvexSeparation.cpp engine/runtime/StarFanSeparation.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-ffp-contract=off")
endif()
add_library(spacerad_services STATIC src/AnalysisService.cpp include/spacerad/AnalysisService.h)
add_library(SpaceRad::Services ALIAS spacerad_services)
set_target_properties(spacerad_services PROPERTIES EXPORT_NAME Services)
target_link_libraries(spacerad_services PUBLIC spacerad_io)
if(SPACERAD_BUILD_WORKER)
 find_package(Geant4 11.3 REQUIRED CONFIG)
 file(GLOB _transport CONFIGURE_DEPENDS engine/transport/*.cpp)
 add_executable(spacerad_worker engine/RadiationWorker.cpp ${_transport} engine/runtime/RadiationRuntime.cpp engine/runtime/BatchRunner.cpp engine/runtime/ShieldingScanRunner.cpp engine/runtime/ResponseCacheRunner.cpp)
 set_target_properties(spacerad_worker PROPERTIES OUTPUT_NAME scdat_radiation_worker AUTOMOC OFF)
 target_include_directories(spacerad_worker PRIVATE ${Geant4_INCLUDE_DIRS})
 target_link_libraries(spacerad_worker PRIVATE spacerad_io ${Geant4_LIBRARIES})
 set(_manifest "")
 foreach(_dataset IN LISTS Geant4_DATASETS)
  get_filename_component(_name "${Geant4_DATASET_${_dataset}_PATH}" NAME)
  string(APPEND _manifest "${Geant4_DATASET_${_dataset}_ENVVAR}|${_name}\n")
 endforeach()
 set(_context "{\"schema\":\"spacerad.worker-build/1\",\"compiled_geant4_version\":\"${Geant4_VERSION}\",\"worker_role\":\"standalone\",\"application_version\":\"${PROJECT_VERSION}\",\"paths\":{\"qt_prefix\":\".\",\"worker_directory\":\".\"}}\n")
 string(SHA256 _context_sha "${_context}")
 string(SHA256 _manifest_sha "${_manifest}")
 target_compile_definitions(spacerad_worker PRIVATE SCDAT_BUILD_CONTEXT_SHA256="${_context_sha}" SCDAT_DATASET_MANIFEST_SHA256="${_manifest_sha}")
 file(GENERATE OUTPUT "$<TARGET_FILE_DIR:spacerad_worker>/radiation-build.json" CONTENT "${_context}" NEWLINE_STYLE UNIX)
 file(GENERATE OUTPUT "$<TARGET_FILE_DIR:spacerad_worker>/radiation-datasets.txt" CONTENT "${_manifest}" NEWLINE_STYLE UNIX)
 install(TARGETS spacerad_worker RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 install(FILES "$<TARGET_FILE_DIR:spacerad_worker>/radiation-build.json" "$<TARGET_FILE_DIR:spacerad_worker>/radiation-datasets.txt" DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(SPACERAD_BUILD_DESKTOP)
 find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets)
 find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel FiltersSources RenderingCore RenderingOpenGL2 InteractionStyle GUISupportQt)
 add_library(spacerad_widgets STATIC src/WorkbenchWidget.cpp src/GeometryView.cpp include/spacerad/WorkbenchWidget.h src/GeometryView.h)
 add_library(SpaceRad::Widgets ALIAS spacerad_widgets)
 set_target_properties(spacerad_widgets PROPERTIES EXPORT_NAME Widgets)
 target_link_libraries(spacerad_widgets PUBLIC spacerad_services Qt6::Widgets PRIVATE Qt6::OpenGLWidgets ${VTK_LIBRARIES})
 vtk_module_autoinit(TARGETS spacerad_widgets MODULES ${VTK_LIBRARIES})
 add_library(spacerad_shell STATIC app/HostMain.cpp)
 target_include_directories(spacerad_shell PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/app)
 target_link_libraries(spacerad_shell PUBLIC spacerad_widgets PRIVATE Qt6::OpenGLWidgets ${VTK_LIBRARIES})
 add_executable(spacerad WIN32 app/main.cpp)
 target_link_libraries(spacerad PRIVATE spacerad_shell)
 add_executable(spacerad_embedded_example WIN32 examples/embedded/main.cpp)
 target_link_libraries(spacerad_embedded_example PRIVATE spacerad_shell)
 vtk_module_autoinit(TARGETS spacerad spacerad_embedded_example MODULES ${VTK_LIBRARIES})
 install(TARGETS spacerad spacerad_embedded_example RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME independent_layout COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_project_layout.py")
 file(GLOB _tests CONFIGURE_DEPENDS tests/engine/test_*.cpp)
 foreach(_file IN LISTS _tests)
  get_filename_component(_name "${_file}" NAME_WE)
  add_executable(${_name} "${_file}")
  set_target_properties(${_name} PROPERTIES AUTOMOC OFF)
  target_link_libraries(${_name} PRIVATE spacerad_io)
  add_test(NAME ${_name} COMMAND ${_name})
 endforeach()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_service_runtime.cpp")
  add_executable(test_service_runtime tests/test_service_runtime.cpp)
  target_link_libraries(test_service_runtime PRIVATE spacerad_services)
 endif()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_service.cpp")
  add_executable(test_service tests/test_service.cpp)
  target_link_libraries(test_service PRIVATE spacerad_services)
  add_test(NAME service_contract COMMAND test_service)
 endif()
endif()

include(CMakePackageConfigHelpers)
install(TARGETS spacerad_json spacerad_model spacerad_io spacerad_services EXPORT SpaceRadTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} INCLUDES DESTINATION include)
if(TARGET spacerad_widgets)
 install(TARGETS spacerad_widgets EXPORT SpaceRadTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/spacerad DESTINATION include)
install(DIRECTORY engine/ DESTINATION include/spacerad/engine FILES_MATCHING PATTERN "*.h" PATTERN "RadiationProject.h" EXCLUDE)
install(DIRECTORY third_party/nlohmann_json/include/nlohmann DESTINATION include)
install(EXPORT SpaceRadTargets NAMESPACE SpaceRad:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpaceRad)
configure_package_config_file(cmake/SpaceRadConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SpaceRadConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpaceRad)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SpaceRadConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SpaceRadConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/SpaceRadConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpaceRad)
