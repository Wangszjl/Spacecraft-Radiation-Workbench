cmake_minimum_required(VERSION 3.22)
project(SpaceRadConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SpaceRad 0.1 REQUIRED CONFIG)
add_executable(spacerad_package_consumer main.cpp)
if(LEGACY_RADIATION_INCLUDE)
 target_include_directories(spacerad_package_consumer BEFORE PRIVATE "${LEGACY_RADIATION_INCLUDE}")
 target_compile_definitions(spacerad_package_consumer PRIVATE TEST_LEGACY_HEADER=1)
endif()
target_link_libraries(spacerad_package_consumer PRIVATE SpaceRad::Services)
