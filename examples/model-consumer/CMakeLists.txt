cmake_minimum_required(VERSION 3.22)
project(SpaceRadModelConsumer LANGUAGES CXX)
include(CTest)
find_package(SpaceRad 0.1 CONFIG REQUIRED COMPONENTS Model)
if(TARGET SpaceRad::Services OR TARGET SpaceRad::Widgets)
 message(FATAL_ERROR "This acceptance example expects the model-only SDK")
endif()
add_executable(spacerad_model_consumer main.cpp)
target_link_libraries(spacerad_model_consumer PRIVATE SpaceRad::Model)
add_test(NAME installed_model_consumer COMMAND spacerad_model_consumer)
